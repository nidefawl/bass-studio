cmake_minimum_required(VERSION 3.3)
message (STATUS "CMake version: ${CMAKE_VERSION}")

set(DAW_SRC_PATH "${CMAKE_SOURCE_DIR}/src")
set(BUILD_NAME "DAW")
project(${BUILD_NAME} C CXX)
project("vstscanner" C CXX)
option(BUILD_TESTS "Build tests" OFF)

set_property(GLOBAL PROPERTY RULE_LAUNCH_COMPILE "${CMAKE_COMMAND} -E time")
include("setup.cmake")
include("cotire.cmake")
add_definitions("-DUSE_WIN32_EXC_HOOKS")

# option(BUILD_HOST "Build daw" ON)
# option(BUILD_SCANNER "Build plugin scanner" OFF)

file(GLOB_RECURSE HDR_ALL_FILES src/*.h)
file(GLOB_RECURSE CPP_ALL_FILES src/*.cpp)
file(GLOB_RECURSE C_SRC_FILES src/*.c)
list(APPEND HDR_FILES "")
list(APPEND CPP_SRC_FILES "")
foreach(f ${HDR_ALL_FILES})
    if (f MATCHES ".*/tests/[^/]*$")
        continue()
    endif()
    if (UNIX)
      if (f MATCHES ".*/host/platform/win/[^/]*$")
          continue()
      endif()
      if (f MATCHES ".*/platform/win/[^/]*$")
          continue()
      endif()
      if (f MATCHES ".*/platform/mingw/[^/]*$")
          continue()
      endif()
      if (f MATCHES ".*/external_asio.cpp$")
          continue()
      endif()
    endif (UNIX)
    if (WIN32)
      if (f MATCHES ".*/host/platform/linux/[^/]*$")
          continue()
      endif()
      if (f MATCHES ".*/platform/linux/[^/]*$")
          continue()
      endif()
    endif (WIN32)
    if (MSVC)
      if (f MATCHES ".*/platform/mingw/[^/]*$")
          continue()
      endif()
    endif (MSVC)
    list(APPEND HDR_FILES ${f})
endforeach()

foreach(f ${CPP_ALL_FILES})
    if (f MATCHES ".*/scanner.cpp$")
        continue()
    endif()
    if (f MATCHES "main.cpp$")
        continue()
    endif()
    if (f MATCHES ".*/tests/[^/]*$")
        continue()
    endif()
    if (UNIX)
	    if (f MATCHES ".*/host/platform/win/[^/]*$")
	        continue()
	    endif()
	    if (f MATCHES ".*/platform/win/[^/]*$")
	        continue()
	    endif()
	    if (f MATCHES ".*/platform/mingw/[^/]*$")
	        continue()
	    endif()
	    if (f MATCHES ".*/external_asio.cpp$")
	        continue()
	    endif()
	  endif (UNIX)
    if (WIN32)
	    if (f MATCHES ".*/host/platform/linux/[^/]*$")
	        continue()
	    endif()
	    if (f MATCHES ".*/platform/linux/[^/]*$")
	        continue()
	    endif()
	  endif (WIN32)
    if (MSVC)
	    if (f MATCHES ".*/platform/mingw/[^/]*$")
	        continue()
	    endif()
	  endif (MSVC)
    list(APPEND CPP_SRC_FILES ${f})
endforeach()

source_group("Sources" FILES ${CPP_SRC_FILES})
source_group("Headers" FILES ${HDR_FILES})

#build a fat library only when tests are enabled
#if tests are disabled we link objects directly to output executable (faster)

if (BUILD_TESTS)
  add_executable(${BUILD_NAME} "src/main.cpp")
  add_library("dawlib" STATIC ${CPP_SRC_FILES} ${C_SRC_FILES} ${HDR_FILES})
  set(LINK_TARGET "dawlib")
  target_link_libraries(${BUILD_NAME} PUBLIC "dawlib")
else()
add_executable(${BUILD_NAME} "src/main.cpp" ${CPP_SRC_FILES} ${C_SRC_FILES} ${HDR_FILES})
set(LINK_TARGET ${BUILD_NAME})
add_executable("vstscanner" "src/host/scanner.cpp" ${CPP_SRC_FILES} ${C_SRC_FILES} ${HDR_FILES})
set(LINK_TARGET_SCANNER "vstscanner")
endif()

include("includes.cmake")
include("libs.cmake")

target_link_libraries(${LINK_TARGET} PRIVATE ${GLFW_LIB} ${SQLITECPP_LIB} ${SOXR_LIB} ${PORTAUDIO_LIB})
target_link_libraries(${LINK_TARGET_SCANNER} PRIVATE ${GLFW_LIB} ${SQLITECPP_LIB} ${SOXR_LIB} ${PORTAUDIO_LIB})

if (UNIX)
target_link_libraries(${LINK_TARGET} PRIVATE ${X11_X11_LIB} ${CMAKE_THREAD_LIBS_INIT} ${CMAKE_DL_LIBS} ${ALSA_LIBRARY} ${GTK3_LIBRARIES} rt m)
target_link_libraries(${LINK_TARGET_SCANNER} PRIVATE ${X11_X11_LIB} ${CMAKE_THREAD_LIBS_INIT} ${CMAKE_DL_LIBS} ${ALSA_LIBRARY} ${GTK3_LIBRARIES} rt m)
endif(UNIX)
if (WIN32)
target_link_libraries(${LINK_TARGET} PRIVATE winmm kernel32 gdi32 ole32 uuid comdlg32 setupapi mswsock ws2_32)
target_link_libraries(${LINK_TARGET_SCANNER} PRIVATE winmm kernel32 gdi32 ole32 uuid comdlg32 setupapi mswsock ws2_32)
endif(WIN32)

set(SCANNER_EXE_NAME "vstscanner-${CMAKE_CXX_COMPILER_ID}")
set_target_properties("vstscanner" PROPERTIES OUTPUT_NAME ${SCANNER_EXE_NAME})
set_target_properties("vstscanner" PROPERTIES RUNTIME_OUTPUT_DIRECTORY ../run/$<$<CONFIG:Debug>:>)
set_target_properties("vstscanner" PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY ../run/)

set_target_properties(${BUILD_NAME} PROPERTIES OUTPUT_NAME ${DAW_EXE_NAME})
set_target_properties(${BUILD_NAME} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ../run/$<$<CONFIG:Debug>:>)
set_target_properties(${BUILD_NAME} PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY ../run/)


set(COTIRE_UNITY_SOURCE_EXCLUDE_EXTENSIONS "m;mm;c")
message(STATUS "COTIRE_UNITY_SOURCE_EXCLUDE_EXTENSIONS: ${COTIRE_UNITY_SOURCE_EXCLUDE_EXTENSIONS}")

set_source_files_properties("${DAW_SRC_PATH}/platform/win/fileio.cpp" PROPERTIES COTIRE_EXCLUDED TRUE)
set_source_files_properties("${DAW_SRC_PATH}/window/window.cpp" PROPERTIES COTIRE_EXCLUDED TRUE)
set_source_files_properties("${DAW_SRC_PATH}/window/debugwindow.cpp" PROPERTIES COTIRE_EXCLUDED TRUE)
